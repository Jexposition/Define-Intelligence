NavierStokes.AxisymmetricFields.velocity_zero (H K : AxisymmetricFields.Profile) (t : ℝ) (x : Space)
  (hH : DifferentiableAt ℝ H (AxisymmetricFields.profilePoint t x))
  (hK : DifferentiableAt ℝ K (AxisymmetricFields.profilePoint t x)) :
  (AxisymmetricFields.velocity H K (t, x)).ofLp 0 =
    -x.ofLp 0 * AxisymmetricFields.partialZ H (AxisymmetricFields.profilePoint t x) / 2 +
      x.ofLp 1 * AxisymmetricFields.partialS K (AxisymmetricFields.profilePoint t x)
NavierStokes.AxisymmetricFields.velocity_one (H K : AxisymmetricFields.Profile) (t : ℝ) (x : Space)
  (hH : DifferentiableAt ℝ H (AxisymmetricFields.profilePoint t x))
  (hK : DifferentiableAt ℝ K (AxisymmetricFields.profilePoint t x)) :
  (AxisymmetricFields.velocity H K (t, x)).ofLp 1 =
    -x.ofLp 1 * AxisymmetricFields.partialZ H (AxisymmetricFields.profilePoint t x) / 2 -
      x.ofLp 0 * AxisymmetricFields.partialS K (AxisymmetricFields.profilePoint t x)
NavierStokes.AxisymmetricFields.velocity_two (H K : AxisymmetricFields.Profile) (t : ℝ) (x : Space)
  (hH : DifferentiableAt ℝ H (AxisymmetricFields.profilePoint t x))
  (hK : DifferentiableAt ℝ K (AxisymmetricFields.profilePoint t x)) :
  (AxisymmetricFields.velocity H K (t, x)).ofLp 2 =
    H (AxisymmetricFields.profilePoint t x) +
      AxisymmetricFields.radialEnergy x * AxisymmetricFields.partialS H (AxisymmetricFields.profilePoint t x)
